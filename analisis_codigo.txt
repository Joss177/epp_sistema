"""
Gestión de alertas: cooldown para no saturar WhatsApp, guardado de evidencia
en disco, registro en base de datos SQLite, y envío por GreenAPI.
"""

import cv2
import requests
import sqlite3
import time
import os
import logging
from datetime import datetime

logger = logging.getLogger(__name__)


class AlertManager:
    def __init__(self, greenapi_config, storage_config, cooldown_minutes: int = 3):
        self.id_instance = greenapi_config["id_instance"]
        self.api_token = greenapi_config["api_token_instance"]
        self.chat_id = greenapi_config["chat_id"]

        self.snapshots_dir = storage_config["snapshots_dir"]
        self.db_path = storage_config["database_path"]
        self.cooldown_seconds = cooldown_minutes * 60

        os.makedirs(self.snapshots_dir, exist_ok=True)
        os.makedirs(os.path.dirname(self.db_path), exist_ok=True)

        self._last_alert_time = {}  # camera_id -> timestamp del último aviso
        self._init_db()

    def _init_db(self):
        conn = sqlite3.connect(self.db_path)
        conn.execute("""
            CREATE TABLE IF NOT EXISTS eventos (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                camara_id TEXT NOT NULL,
                epp_faltante TEXT NOT NULL,
                fecha_hora TEXT NOT NULL,
                snapshot_path TEXT
            )
        """)
        conn.commit()
        conn.close()

    def _en_cooldown(self, camera_id: str) -> bool:
        ultimo = self._last_alert_time.get(camera_id, 0)
        return (time.time() - ultimo) < self.cooldown_seconds

    def _guardar_snapshot(self, camera_id: str, frame) -> str:
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"{camera_id}_{timestamp}.jpg"
        filepath = os.path.join(self.snapshots_dir, filename)
        cv2.imwrite(filepath, frame)
        return filepath

    def _registrar_evento(self, camera_id: str, epp_faltante: list, snapshot_path: str):
        conn = sqlite3.connect(self.db_path)
        conn.execute(
            "INSERT INTO eventos (camara_id, epp_faltante, fecha_hora, snapshot_path) VALUES (?, ?, ?, ?)",
            (camera_id, ", ".join(epp_faltante), datetime.now().isoformat(), snapshot_path),
        )
        conn.commit()
        conn.close()

    def _enviar_whatsapp(self, mensaje: str, snapshot_path: str):
        url = f"https://api.greenapi.com/waInstance{self.id_instance}/sendFileByUpload/{self.api_token}"

        with open(snapshot_path, "rb") as f:
            files = {"file": (os.path.basename(snapshot_path), f, "image/jpeg")}
            data = {"chatId": self.chat_id, "caption": mensaje}
            try:
                response = requests.post(url, data=data, files=files, timeout=15)
                response.raise_for_status()
                logger.info(f"Alerta enviada por WhatsApp: {response.status_code}")
            except requests.RequestException as e:
                logger.error(f"Error enviando alerta a GreenAPI: {e}")

    def procesar_incumplimiento(self, camera_id: str, epp_faltante: list, frame):
        """
        Punto de entrada principal: decide si corresponde alertar (respetando
        cooldown), guarda evidencia, registra el evento y notifica por WhatsApp.
        """
        if self._en_cooldown(camera_id):
            return

        snapshot_path = self._guardar_snapshot(camera_id, frame)
        self._registrar_evento(camera_id, epp_faltante, snapshot_path)

        mensaje = (
            f"⚠️ *Alerta de EPP incompleto*\n"
            f"Cámara: {camera_id}\n"
            f"Falta: {', '.join(epp_faltante)}\n"
            f"Hora: {datetime.now().strftime('%H:%M:%S %d/%m/%Y')}"
        )
        self._enviar_whatsapp(mensaje, snapshot_path)

        self._last_alert_time[camera_id] = time.time()

"""
Lectura de cámara RTSP en un hilo separado.

Por qué un hilo aparte: OpenCV va llenando un buffer interno con frames de la
cámara. Si tu CPU tarda en procesar cada frame (por el modelo de detección),
el buffer se acumula y terminas viendo/analizando video "atrasado" respecto
al tiempo real. Con un hilo dedicado a leer, siempre nos quedamos con el
frame MÁS RECIENTE y descartamos los viejos.
"""

import cv2
import threading
import time
import logging

logger = logging.getLogger(__name__)


class CameraStream:
    def __init__(self, camera_id: str, rtsp_url: str, reconnect_delay: int = 5):
        self.camera_id = camera_id
        self.rtsp_url = rtsp_url
        self.reconnect_delay = reconnect_delay

        self.capture = None
        self.latest_frame = None
        self.lock = threading.Lock()
        self.running = False
        self.thread = None

    def start(self):
        self.running = True
        self.thread = threading.Thread(target=self._update_loop, daemon=True)
        self.thread.start()
        logger.info(f"[{self.camera_id}] Hilo de captura iniciado")
        return self

    def _connect(self):
        capture = cv2.VideoCapture(self.rtsp_url, cv2.CAP_FFMPEG)
        # Buffer chico: minimiza el delay entre lo que pasa y lo que vemos
        capture.set(cv2.CAP_PROP_BUFFERSIZE, 1)
        return capture

    def _update_loop(self):
        self.capture = self._connect()

        while self.running:
            if self.capture is None or not self.capture.isOpened():
                logger.warning(f"[{self.camera_id}] Sin conexión, reintentando en {self.reconnect_delay}s...")
                time.sleep(self.reconnect_delay)
                self.capture = self._connect()
                continue

            ok, frame = self.capture.read()
            if not ok:
                logger.warning(f"[{self.camera_id}] Frame no leído, reconectando...")
                self.capture.release()
                self.capture = None
                time.sleep(self.reconnect_delay)
                continue

            with self.lock:
                self.latest_frame = frame

    def read(self):
        """Devuelve el frame más reciente disponible (o None si aún no hay ninguno)."""
        with self.lock:
            return None if self.latest_frame is None else self.latest_frame.copy()

    def stop(self):
        self.running = False
        if self.thread:
            self.thread.join(timeout=2)
        if self.capture:
            self.capture.release()
        logger.info(f"[{self.camera_id}] Detenido")

"""
Orquestador principal: lee cámaras, corre detección de EPP, y dispara
alertas cuando corresponde.

Uso:
    python src/main.py
"""

import yaml
import time
import logging
from collections import defaultdict

from camera_stream import CameraStream
from ppe_detector import PPEDetector
from alert_manager import AlertManager

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
)
logger = logging.getLogger(__name__)


def cargar_config(path="config.yaml"):
    with open(path, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def main():
    config = cargar_config()

    # --- Iniciar cámaras ---
    streams = {}
    for cam_cfg in config["cameras"]:
        stream = CameraStream(cam_cfg["id"], cam_cfg["rtsp_url"]).start()
        streams[cam_cfg["id"]] = stream

    # --- Cargar detector ---
    det_cfg = config["detection"]
    detector = PPEDetector(
        model_path=det_cfg["model_path"],
        confidence_threshold=det_cfg["confidence_threshold"],
        resize_width=det_cfg["resize_width"],
    )

    # --- Gestor de alertas ---
    alert_manager = AlertManager(
        greenapi_config=config["greenapi"],
        storage_config=config["storage"],
        cooldown_minutes=det_cfg["alert_cooldown_minutes"],
    )

    frame_skip = det_cfg["frame_skip"]
    seconds_before_alert = det_cfg["seconds_before_alert"]
    required_ppe = det_cfg["required_ppe"]

    # Trackea desde cuándo cada cámara viene detectando incumplimiento sostenido
    incumplimiento_desde = defaultdict(lambda: None)

    logger.info("Sistema iniciado. Esperando frames de las cámaras...")

    frame_counter = 0
    try:
        while True:
            frame_counter += 1
            time.sleep(0.1)  # ritmo del loop principal

            if frame_counter % frame_skip != 0:
                continue

            for camera_id, stream in streams.items():
                frame = stream.read()
                if frame is None:
                    continue  # cámara aún conectando

                detections = detector.detect(frame)
                faltante = detector.evaluar_cumplimiento(detections, required_ppe)

                if faltante:
                    if incumplimiento_desde[camera_id] is None:
                        incumplimiento_desde[camera_id] = time.time()

                    tiempo_incumpliendo = time.time() - incumplimiento_desde[camera_id]
                    if tiempo_incumpliendo >= seconds_before_alert:
                        logger.warning(f"[{camera_id}] EPP incompleto sostenido: {faltante}")
                        alert_manager.procesar_incumplimiento(camera_id, faltante, frame)
                        incumplimiento_desde[camera_id] = None  # reinicia el contador tras alertar
                else:
                    incumplimiento_desde[camera_id] = None  # se cumplió, reinicia

    except KeyboardInterrupt:
        logger.info("Deteniendo sistema...")
    finally:
        for stream in streams.values():
            stream.stop()


if __name__ == "__main__":
    main()

"""
Detección de personas y EPP (casco, botas, guantes) usando YOLO.

IMPORTANTE: el modelo YOLO estándar (entrenado con COCO) NO detecta cascos,
botas ni guantes — solo detecta "person" y objetos genéricos. Para que esto
funcione necesitas un modelo entrenado específicamente para EPP (ver README,
sección "Conseguir un modelo de EPP").

Este módulo asume que tu modelo .pt tiene clases con nombres como:
  "casco", "sin_casco", "botas", "sin_botas", "guantes", "sin_guantes"
(o en inglés: "helmet", "no_helmet", "boots", "gloves", etc.)

Ajusta CLASS_NAME_MAP abajo según las clases reales de tu modelo.
"""

from ultralytics import YOLO
import logging

logger = logging.getLogger(__name__)

# Mapea el nombre de clase que devuelve TU modelo -> nombre estándar interno.
# Edita esto según lo que imprima `model.names` (ver comentario al final del archivo).
CLASS_NAME_MAP = {
    "helmet": "casco",
    "no_helmet": "sin_casco",
    "boots": "botas",
    "no_boots": "sin_botas",
    "gloves": "guantes",
    "no_gloves": "sin_guantes",
    "person": "persona",
}


class PPEDetector:
    def __init__(self, model_path: str, confidence_threshold: float = 0.5, resize_width: int = 640):
        logger.info(f"Cargando modelo desde {model_path}...")
        self.model = YOLO(model_path)
        self.confidence_threshold = confidence_threshold
        self.resize_width = resize_width
        logger.info(f"Modelo cargado. Clases disponibles: {self.model.names}")

    def detect(self, frame):
        """
        Corre la detección sobre un frame.
        Devuelve una lista de detecciones: [{"clase": str, "confianza": float, "box": (x1,y1,x2,y2)}, ...]
        """
        # Reescalar para acelerar inferencia en CPU
        h, w = frame.shape[:2]
        if w > self.resize_width:
            scale = self.resize_width / w
            frame = cv2_resize(frame, self.resize_width, int(h * scale))

        results = self.model.predict(
            frame,
            conf=self.confidence_threshold,
            verbose=False,
            device="cpu",
        )

        detections = []
        for result in results:
            for box in result.boxes:
                class_id = int(box.cls[0])
                raw_name = self.model.names[class_id]
                mapped_name = CLASS_NAME_MAP.get(raw_name, raw_name)
                confidence = float(box.conf[0])
                xyxy = box.xyxy[0].tolist()
                detections.append({
                    "clase": mapped_name,
                    "confianza": confidence,
                    "box": xyxy,
                })

        return detections

    def evaluar_cumplimiento(self, detections, required_ppe):
        """
        A partir de las detecciones de un frame, determina qué EPP requerido
        NO fue detectado. Devuelve una lista de EPP faltante, ej: ["botas", "guantes"]

        Nota: esta es una versión simple que asume una persona por frame.
        Si necesitas trackear múltiples personas por separado en el mismo
        frame, hay que agregar asociación de boxes por persona (te ayudo si
        llegas a necesitarlo).
        """
        clases_detectadas = {d["clase"] for d in detections}
        faltante = [item for item in required_ppe if item not in clases_detectadas]
        return faltante


def cv2_resize(frame, width, height):
    import cv2
    return cv2.resize(frame, (width, height))

def dibujar_ui(frame, preds, faltantes, cam_nombre):
    # Bounding boxes
    for pred in preds:
        clase  = pred["class"].lower()
        umbral = CONFIANZA_POR_CLASE.get(clase, 0.3)
        if pred["confidence"] < umbral:
            continue
        x = int(pred["x"] - pred["width"]  / 2)

        w = int(pred["width"])
        h = int(pred["height"])
        cv2.rectangle(frame, (x, y), (x+w, y+h), (0, 200, 0), 2)

        color = (0, 220, 0) if ok else (0, 60, 220)

    # Nombre de camara
    h_frame, w_frame = frame.shape[:2]

# Umbral de confianza por clase (más bajo = más fácil de detectar)
CONFIANZA_POR_CLASE = {
    "hardhat": 0.5,   # Casco detecta bien, umbral normal
    "gloves":  0.15,  # Guantes detecta menos, umbral más bajo
    "goggles": 0.15,  # Careta detecta menos, umbral más bajo
}

def detectar_faltantes(preds):
    detectados = set()
    for pred in preds:
        clase = pred["class"].lower()
        umbral = CONFIANZA_POR_CLASE.get(clase, 0.3)
        if pred["confidence"] >= umbral:
            detectados.add(clase)
    return [nombre for clave, nombre in EPP_REQUERIDO.items()
            if clave not in detectados]

def dibujar_ui(frame, preds, faltantes):
    for pred in preds:
        if pred["confidence"] < 0.3:
            continue
        x = int(pred["x"] - pred["width"]  / 2)
        y = int(pred["y"] - pred["height"] / 2)
        w = int(pred["width"])
        h = int(pred["height"])
        cv2.rectangle(frame, (x, y), (x+w, y+h), (0, 200, 0), 2)
        cv2.putText(frame, pred["class"] + " " + str(round(pred["confidence"]*100)) + "%",
                    (x, y - 8), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (0, 200, 0), 2)

    fondo_h = len(EPP_REQUERIDO) * 30 + 10
    cv2.rectangle(frame, (0, 0), (220, fondo_h), (30, 30, 30), -1)
    for i, (clave, nombre) in enumerate(EPP_REQUERIDO.items()):
        ok    = nombre not in faltantes
        color = (0, 220, 0) if ok else (0, 60, 220)
        icono = "OK" if ok else "!!"
        cv2.putText(frame, "[" + icono + "] " + nombre,
                    (8, 25 + i * 30), cv2.FONT_HERSHEY_SIMPLEX, 0.65, color, 2)

    ts = datetime.datetime.now().strftime("%d/%m/%Y %H:%M:%S")
    cv2.putText(frame, ts, (10, frame.shape[0] - 10),
                cv2.FONT_HERSHEY_SIMPLEX, 0.55, (200, 200, 200), 1)

    if faltantes:
        h, w = frame.shape[:2]
        cv2.rectangle(frame, (0, h - 45), (w, h - 25), (0, 0, 180), -1)
        cv2.putText(frame, "FALTA: " + ", ".join(faltantes),
                    (10, h - 30), cv2.FONT_HERSHEY_SIMPLEX, 0.75, (255, 255, 255), 2)
    return frame

    data = r.json()

        )
        requests.delete(del_url, timeout=10)
